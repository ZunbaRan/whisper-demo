async function loadFeeds() {
    try {
        const stats = await callApi('/db/stats');
        feeds = stats.feeds || [];
        
        const feedFilter = document.getElementById('feed-filter');
        feeds.forEach(feed => {
            const option = document.createElement('option');
            option.value = feed.title;
            option.textContent = `${feed.title} (${feed.count})`;
            feedFilter.appendChild(option);
        });
    } catch (error) {
        console.error('加载 Feed 失败:', error);
    }
}

async function loadEntries(page = 1, feed_title = "", search_value = "") {
    try {
        const limit = 20;
        const offset = (page - 1) * limit;

        let url = `/db/entries?limit=${limit}&offset=${offset}&count=true`;
        if (feed_title) {
            url += `&feed_title=${encodeURIComponent(feed_title)}`;
        }
        if (search_value) {
            url += `&search_value=${encodeURIComponent(search_value)}`;
        }

        const response = await callApi(url);
        const entries = response.entries;
        const totalCount = response.total;
        
        const tableBody = document.getElementById('entries-table');
        
        if (entries.length === 0) {
            tableBody.innerHTML = '没有找到条目';
            return;
        }
        
        let html = '';
        entries.forEach(entry => {
            // 处理可能为空的字段
            const title = entry.title || '';
            const summary = entry.summary || '';
            const description = entry.description || '';
            const feed_title = entry.feed_title || '';
            
            // 创建可复制的单元格内容
            const createCopyableCell = (content, maxLength = 100) => {
                const shortContent = content.length > maxLength ? content.substring(0, maxLength) + '...' : content;
                return `
                    
                        ${shortContent}
                    
                `;
            };
            
            html += `
                
                    ${entry.id}
                    ${createCopyableCell(title)}
                    ${createCopyableCell(summary)}
                    ${createCopyableCell(description)}
                    ${entry.publishedAt || ''}
                    ${feed_title}
                    
                        
                            ${getStatusIcon(entry)}
                        
                    
                    
                        

                            
                            
                        

                    
                
            `;
        });
        
        tableBody.innerHTML = html;
        
        // 初始化工具提示 (Bootstrap 5)
        var tooltipTriggerList = [].slice.call(document.querySelectorAll('[data-bs-toggle="tooltip"]'))
        var tooltipList = tooltipTriggerList.map(function (tooltipTriggerEl) {
            return new bootstrap.Tooltip(tooltipTriggerEl, {
                trigger: 'hover',
                html: true,
                delay: { show: 300, hide: 500 },  // 增加隐藏延迟到 2000ms (2秒)
                template: '
'
            });
        });
        
        // 更新分页，使用返回的总条目数
        updatePaginationWithTotal(page, totalCount);
    } catch (error) {
        console.error('加载条目失败:', error);
        document.getElementById('entries-table').innerHTML = `
            加载失败: ${error.message}
        `;
    }
}

function copyToClipboard(text, element) {
    const textarea = document.createElement('textarea');
    textarea.value = text;
    document.body.appendChild(textarea);
    textarea.select();
    document.execCommand('copy');
    document.body.removeChild(textarea);
    
    // 使用 Bootstrap Toast
    const toastEl = document.getElementById('copyToast');
    const toast = new bootstrap.Toast(toastEl);
    toast.show();
}

function formatDate(dateString) {
    if (!dateString) return '-';
    
    try {
        const date = new Date(dateString);
        return date.toLocaleString('zh-CN');
    } catch (e) {
        return dateString;
    }
}

async function processEntry(entryId) {
    try {
        // 显示进度模态框
        const progressModal = new bootstrap.Modal(document.getElementById('progressModal'));
        progressModal.show();
        
        // 更新进度信息
        document.getElementById('progress-message').textContent = '正在处理条目...';
        document.getElementById('progress-details').innerHTML = '';
        
        // 调用处理接口
        const response = await fetch(`/workflow/entry/${entryId}`, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            }
        });
        
        if (!response.ok) {
            throw new Error(`处理失败: ${response.statusText}`);
        }
        
        const result = await response.json();
        
        // 显示处理结果
        let detailsHtml = '
    ';
        result.steps.forEach(step => {
            const statusClass = step.success ? 'success' : 'error';
            const statusText = step.success ? 
                (step.skipped ? '已跳过' : '成功') : 
                `失败: ${step.error}`;
            
            detailsHtml += `
                
    
                    ${step.step === 'download' ? '下载' : '转写'}: ${statusText}
                
    
            `;
        });
        detailsHtml += '
    ';
        
        document.getElementById('progress-details').innerHTML = detailsHtml;
        document.getElementById('progress-message').textContent = 
            result.success ? '处理完成' : '处理失败';

        // 3秒后刷新数据
        setTimeout(() => {
            loadEntries(currentPage, result.feedTitle);

// 修改 clean_filename 函数以匹配 utils/file_utils.py 的处理方式
function clean_filename(title) {
    if (!title) {
        return "untitled";
    }
    
    // 替换 Windows 文件名中不允许的字符
    let clean_title = title.replace(/[<>:"\/\\|?*]/g, '_');
    
    // 移除前后空格
    clean_title = clean_title.trim();
    
    // 限制长度
    if (clean_title.length > 200) {
        clean_title = clean_title.substring(0, 200);
    }
    
    // 确保文件名不为空
    if (!clean_title) {
        clean_title = "untitled";
    }
    
    return clean_title;
}

async function viewTranscription(title, entryId) {
    try {
        // 直接使用 entry_id 作为文件名
        const fileName = `${entryId}.txt`;
        
        const response = await fetch(`/output/${fileName}`, {
            method: 'GET',
            headers: {
                'Accept': 'text/plain'
            }
        });
        
        if (!response.ok) {
            throw new Error('转写文件不存在');
        }
        
        const content = await response.text();
        
        // 更新模态框标题和内容
        document.getElementById('transcriptionModalTitle').textContent = title;
        document.getElementById('transcriptionContent').textContent = content;
        
        const transcriptionModal = new bootstrap.Modal(document.getElementById('transcriptionModal'));
        transcriptionModal.show();
    } catch (error) {
        console.error('加载转写内容失败:', error);
        alert('加载转写内容失败: ' + error.message);
    }
}

function copyTranscriptionContent() {
    const content = document.getElementById('transcriptionContent').textContent;
    navigator.clipboard.writeText(content).then(() => {
        // 显示复制成功提示
        alert('内容已复制到剪贴板');
    }).catch(err => {
        console.error('复制失败:', err);
        alert('复制失败');
    });
}

async function generate(entryId) {
    try {
        // 构造请求数据

        let url = `/article-style-create/article-create-style?podcast_file=${encodeURIComponent(entryId)}`;

         const response = await fetch(url, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            }
        });

        if (!response.ok) {
            throw new Error(`生成失败: ${response.statusText}`);
        }

        const result = await response.json();

        // 显示生成结果
        showGenerateResult(entryId, result);
    } catch (error) {
        console.error('生成失败:', error);
        alert('生成失败: ' + error.message);
    }
}

function showGenerateResult(entryId, result) {
    // 构建结果内容
    let resultContent = `
    生成结果 - ${entryId}
    `;

    result.results.forEach((item, index) => {
        resultContent += `
            
    
                
    第 ${index + 1} 部分
    
                
    ${JSON.stringify(item, null, 2)}
    
            
    
        `;
    });

    resultContent += '
    ';

    // 更新模态框内容
    document.getElementById('transcriptionModalTitle').innerHTML = '生成结果';
    document.getElementById('transcriptionContent').innerHTML = resultContent;

    // 显示模态框
    const transcriptionModal = new bootstrap.Modal(document.getElementById('transcriptionModal'));
    transcriptionModal.show();
}
